cmake_minimum_required(VERSION 3.20)
project(subplane VERSION 2.0.0 LANGUAGES C CXX)

# INV-007: C++20, GCC 11 on Ubuntu 22.04 is the target toolchain.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# -O2 over -O3 by decision (see .subplane/decisions.md).
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0")
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g -DNDEBUG")

option(SUBPLANE_BUILD_TESTS "Build unit and integration tests" ON)
option(SUBPLANE_BUILD_TOOLS "Build the subplane CLI (probe, tail, schema, send, replay, dbn-sim, aeron-pub, soak)" ON)
option(SUBPLANE_WERROR "Treat warnings as errors" ON)
option(SUBPLANE_WITH_AERON "Build the aeron transport (fetches the Aeron C client + media driver)" ON)

find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(YAML_CPP REQUIRED IMPORTED_TARGET yaml-cpp)

# Common warnings for all first-party targets.
add_library(subplane_warnings INTERFACE)
target_compile_options(subplane_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion)
if(SUBPLANE_WERROR)
    target_compile_options(subplane_warnings INTERFACE -Werror)
endif()

# Vendored third-party (no warnings enforced).
add_library(pugixml STATIC third_party/pugixml/pugixml.cpp)
target_include_directories(pugixml PUBLIC third_party/pugixml)
set_target_properties(pugixml PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(doctest INTERFACE)
target_include_directories(doctest INTERFACE third_party/doctest)

if(SUBPLANE_WITH_AERON)
    include(cmake/aeron.cmake)
endif()

# Core library: everything except main().
add_library(subplane_core STATIC
    src/agent.cpp
    src/api_server.cpp
    src/config_manager.cpp
    src/logging/log_registry.cpp
    src/metrics/metrics_manager.cpp
    src/persistence_manager.cpp
    src/symbol_table.cpp
    src/core/filter_table.cpp
    src/decoder/sbe_schema.cpp
    src/decoder/sbe_routing_decoder.cpp
    src/decoder/dbn_decoder.cpp
    src/decoder/flatbuffers_decoder.cpp
    src/decoder/sbe_views.cpp
    src/decoder/sbe_codec.cpp
    src/decoder/sbe_lint.cpp
    src/framing/sbe_length.cpp
    src/framing/framing.cpp
    src/transport/udp_multicast.cpp
    src/transport/tcp_stream.cpp
    src/recap/recap_config.cpp
    src/recap/recap_manager.cpp
    src/recap/recap_status.cpp
    src/registry/builtin_components.cpp
    src/pipeline/session.cpp
    src/pipeline/partition.cpp
    src/pipeline/lvc.cpp
    src/pipeline/pipeline.cpp
    src/pipeline/recap_state.cpp
    src/pipeline/session_manager.cpp
    src/control/control_server.cpp
)
target_include_directories(subplane_core PUBLIC include)
target_link_libraries(subplane_core
    PUBLIC PkgConfig::YAML_CPP Threads::Threads
    PRIVATE subplane_warnings pugixml
)
if(SUBPLANE_WITH_AERON)
    target_sources(subplane_core PRIVATE src/transport/aeron_transport.cpp src/transport/aeron_util.cpp src/recap/aeron_recap.cpp)
    target_link_libraries(subplane_core PUBLIC subplane_aeron)
endif()

add_executable(subplaned src/main.cpp)
target_link_libraries(subplaned PRIVATE subplane_core subplane_warnings)
install(TARGETS subplaned DESTINATION bin)

if(SUBPLANE_BUILD_TOOLS AND EXISTS ${CMAKE_SOURCE_DIR}/tools/CMakeLists.txt)
    add_subdirectory(schemas)   # typed views generated by subplane schema at build time
endif()

if(SUBPLANE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

if(EXISTS ${CMAKE_SOURCE_DIR}/examples/CMakeLists.txt)
    add_subdirectory(examples)
endif()

if(SUBPLANE_BUILD_TOOLS AND EXISTS ${CMAKE_SOURCE_DIR}/tools/CMakeLists.txt)
    add_subdirectory(tools)
endif()
